install_dependencies() {
  local project_dir="$1"
  local attempt backup_dir
  for attempt in 1 2 3; do
    if (cd "$project_dir" && npm ci --no-audit --no-fund && npm ls --depth=0); then
      echo "npm ci (${project_dir}) succeeded on attempt ${attempt}"
      return 0
    fi
    echo "::warning::npm ci (${project_dir}) attempt ${attempt} failed or left an inconsistent tree"
    if [ -d "$project_dir/node_modules" ]; then
      backup_dir="$(mktemp -d "${RUNNER_TEMP:-/tmp}/anonyproof-ci-install.XXXXXX")"
      mv "$project_dir/node_modules" "$backup_dir/node_modules"
    fi
    npm cache clean --force || true
  done
  echo "::error::npm ci (${project_dir}) failed after 3 attempts"
  return 1
}
